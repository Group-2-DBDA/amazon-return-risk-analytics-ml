SELECT
    m.parent_asin,
    m.title,
    m.main_category,
    m.category,
    m.brand,
    m.manufacturer,
    m.store,
    m.price,
    m.average_rating,
    m.rating_number,

    -- Robust date parsing (handles bad Amazon metadata)
    -- Robust date parsing (handles bad Amazon metadata)
    date_format(
        COALESCE(
            try(date_parse(m.date_first_available, '%M %d, %Y')),  -- January 29, 2015
            try(date_parse(m.date_first_available, '%d-%b-%y')),   -- 23-Oct-22
            try(date_parse(m.date_first_available, '%Y'))          -- 2018
        ),
        '%d-%m-%Y'
    ) AS date_first_available


    k.total_reviews,
    k.avg_rating,
    k.rating_volatility,
    k.verified_reviews,
    k.unverified_reviews,
    k.verified_5_star_pct,
    k.unverified_5_star_pct,
    k.reviews_with_helpful_votes,
    k.avg_review_length
FROM final_meta22 m
INNER JOIN final_review_kpis k
    ON m.parent_asin = k.parent_asin;
